 const p=projectPayload({tools:[{type:'function',name:'mobility__get_network_status',parameters:{type:'object',properties:{}}}],input:[{type:'function_call',name:'mobility__get_network_status',call_id:'call_real',arguments:'{}'}]},'model_input');

it('audit: stored inspector must publish truncation instead of silently dropping result',async()=>{
 const {safeProjection}=await import('@mobility/contracts');
 const big={tool:'get_incidents',availability:'available',result:{incidents:Array.from({length:100},(_,i)=>({id:String(i),description:'a'.repeat(8000)}))},limitations:['stored only']};
 const projected=safeProjection(big,240000);
 expect(projected.truncated).toBe(true);
 expect(projected.data).not.toHaveProperty('result');
});
